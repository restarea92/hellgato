cmake_minimum_required(VERSION 3.21)
project(HellgatoNativeUi LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(HellgatoNativeUi SHARED native_ui.cpp)
set_target_properties(HellgatoNativeUi PROPERTIES OUTPUT_NAME HellgatoNativeUiV2)
target_compile_features(HellgatoNativeUi PRIVATE cxx_std_17)
target_link_libraries(HellgatoNativeUi PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
target_compile_definitions(HellgatoNativeUi PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(HellgatoNativeUi PRIVATE /utf-8 /Zc:__cplusplus /permissive-)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(HellgatoNativeUiTest test_host.cpp)
    target_compile_features(HellgatoNativeUiTest PRIVATE cxx_std_17)
    target_link_libraries(HellgatoNativeUiTest PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
    target_compile_definitions(HellgatoNativeUiTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(HellgatoNativeUiTest PRIVATE /utf-8 /Zc:__cplusplus /permissive-)
    endif()
    add_test(NAME native_ui_contract COMMAND HellgatoNativeUiTest -platform offscreen)
    add_test(NAME local_dock_reconnect_contract COMMAND HellgatoNativeUiTest -platform offscreen --reconnect-contract)
endif()
